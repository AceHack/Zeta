"""Fitted gloved finger chains: 15 joints per hand, 50 bones per character total."""
import bpy,bmesh,math,json
from pathlib import Path
from mathutils import Vector
R=Path(__file__).resolve().parent;O=R/'blender-v14'
bpy.ops.wm.open_mainfile(filepath=str(O/'Characters-v14.blend'));scene=bpy.context.scene
rows=[]
for name,H in [('AceHack',1.9),('Xenaa',1.82)]:
 rig=bpy.data.objects[name+' | Motion rig'];s=H/1.9
 glove=bpy.data.materials.new(name+' | Fitted finger glove');glove.use_nodes=True;bs=glove.node_tree.nodes['Principled BSDF'];bs.inputs['Base Color'].default_value=(.027,.021,.017,1) if name=='AceHack' else (.017,.011,.025,1);bs.inputs['Roughness'].default_value=.52
 for side,sgn in [('Left',1),('Right',-1)]:
  arm=bpy.data.objects[name+' | '+side+' arm surface'];bm=bmesh.new();bm.from_mesh(arm.data)
  bmesh.ops.bisect_plane(bm,geom=list(bm.verts)+list(bm.edges)+list(bm.faces),dist=1e-6,plane_co=(0,0,1.005*s),plane_no=(0,0,1),clear_inner=True)
  boundary=[e for e in bm.edges if e.is_boundary and all(abs(v.co.z-1.005*s)<1e-5 for v in e.verts)]
  bmesh.ops.holes_fill(bm,edges=boundary,sides=0);bmesh.ops.recalc_face_normals(bm,faces=list(bm.faces));bm.to_mesh(arm.data);bm.free();arm.data.update()
  hand=rig.data.bones[side+'Hand'];wrist=hand.head_local.copy();axis=(hand.tail_local-wrist).normalized();spread=Vector((sgn,0,0));spread=(spread-axis*spread.dot(axis)).normalized();depth=axis.cross(spread).normalized()
  specs=[]
  for finger,lateral,length in [('Index',-.020,.061),('Middle',-.005,.069),('Ring',.010,.064),('Pinky',.023,.050),('Thumb',-.035,.047)]:
   base=wrist+axis*(.052*s if finger!='Thumb' else .020*s)+spread*lateral*s
   direction=(axis+spread*(-.45 if finger=='Thumb' else lateral*2.0)).normalized();length*=s;pts=[base+direction*length*t for t in [0,.39,.73,1]];specs.append((finger,pts,direction))
  bpy.ops.object.select_all(action='DESELECT');rig.select_set(True);bpy.context.view_layer.objects.active=rig;bpy.ops.object.mode_set(mode='EDIT')
  for finger,pts,direction in specs:
   parent=side+'Hand'
   for j in range(3):
    bone=rig.data.edit_bones.new(side+'Hand'+finger+str(j+1));bone.head=pts[j];bone.tail=pts[j+1];bone.parent=rig.data.edit_bones[parent];bone.align_roll(depth);parent=bone.name
  bpy.ops.object.mode_set(mode='OBJECT')
  for finger,pts,direction in specs:
   vs=[];fs=[];ws=[];L=19;N=12;length=(pts[-1]-pts[0]).length
   for j in range(L):
    t=j/(L-1);center=pts[0]+direction*length*t;radius=(.0087 if finger not in ['Pinky','Thumb'] else .0078 if finger=='Pinky' else .010)*s*(.98-.28*t)*(.22+.78*math.sin(min(1,(1-t)*9)*math.pi/2))
    # Small knuckle ridges rather than disconnected cylindrical segments.
    radius*=1+.08*math.exp(-((t-.39)/.06)**2)+.05*math.exp(-((t-.73)/.05)**2)
    for k in range(N):
     a=2*math.pi*k/N;vs.append(tuple(center+spread*radius*math.cos(a)+depth*radius*.83*math.sin(a)))
     blend1=max(0,min(1,(t-.32)/.14));blend2=max(0,min(1,(t-.66)/.14));ws.append({side+'Hand'+finger+'1':1-blend1,side+'Hand'+finger+'2':blend1*(1-blend2),side+'Hand'+finger+'3':blend1*blend2})
   for j in range(L-1):
    for k in range(N):fs.append((j*N+k,j*N+(k+1)%N,(j+1)*N+(k+1)%N,(j+1)*N+k))
   fs.extend([tuple(reversed(range(N))),tuple((L-1)*N+k for k in range(N))])
   me=bpy.data.meshes.new(name+side+finger);me.from_pydata(vs,[],fs);me.update();ob=bpy.data.objects.new(name+' | '+side+' '+finger+' finger',me);scene.collection.objects.link(ob);ob.location.x=rig.location.x;me.materials.append(glove)
   for p in me.polygons:p.use_smooth=True
   groups={n:ob.vertex_groups.new(name=n) for n in ws[0]}
   for i,weights in enumerate(ws):
    for n,w in weights.items():
     if w>1e-8:groups[n].add([i],w,'REPLACE')
   mod=ob.modifiers.new('Finger skeleton','ARMATURE');mod.object=rig;bpy.context.view_layer.update();world=ob.matrix_world.copy();ob.parent=rig;ob.matrix_world=world
 rows.append({'character':name,'bones':len(rig.data.bones),'finger_joints_per_hand':15,'scope':'Blender articulation; Effect House joint mapping not yet tested'})
bpy.ops.wm.save_as_mainfile(filepath=str(O/'Characters-v14.blend'));(R/'finger-rig-v14.json').write_text(json.dumps(rows,indent=2))
