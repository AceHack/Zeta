"""Export one character at a time; mobile images are capped independently of review images."""
import bpy,json
from pathlib import Path
from mathutils import Vector
R=Path(__file__).resolve().parent;O=R/'mobile-export-v14';O.mkdir(exist_ok=True)
bpy.ops.wm.open_mainfile(filepath=str(R/'blender-v14/Characters-v14.blend'));scene=bpy.context.scene;rows=[]
# Explicit neutral defaults and one deterministic image copy per shared texture.
for name in ['AceHack','Xenaa']:
 for side in ['Left','Right']:
  fingers=[ob for ob in scene.objects if ob.name.startswith(name+' | '+side+' ') and ob.name.endswith(' finger')]
  bpy.ops.object.select_all(action='DESELECT')
  for ob in fingers:ob.select_set(True)
  bpy.context.view_layer.objects.active=fingers[0];bpy.ops.object.join();bpy.context.object.name=name+' | '+side+' articulated fingers'
images={};processed_materials=set()
for ob in scene.objects:
 if ob.type!='MESH' or not ob.parent or ob.parent.type!='ARMATURE':continue
 if ob.data.shape_keys:
  for k in ob.data.shape_keys.key_blocks:k.value=0
 for mat in ob.data.materials:
  if not mat or not mat.use_nodes or mat.name in processed_materials:continue
  processed_materials.add(mat.name)
  for node in mat.node_tree.nodes:
   if node.type=='TEX_IMAGE' and node.image:
    src=node.image
    if src.name not in images:
     img=bpy.data.images.new(src.name+' mobile1024',width=src.size[0],height=src.size[1],alpha=False);img.colorspace_settings.name=src.colorspace_settings.name;img.pixels.foreach_set(src.pixels[:])
     if max(img.size)>1024:img.scale(1024,1024)
     img.filepath_raw=str(O/(img.name.replace(' ','-')+'.jpg'));img.file_format='JPEG';scene.render.image_settings.quality=92;img.save();img.pack();images[src.name]=img
    node.image=images[src.name]
  bs=mat.node_tree.nodes.get('Principled BSDF')
  if bs:
   for link in list(bs.inputs['Normal'].links):mat.node_tree.links.remove(link)
for name in ['AceHack','Xenaa']:
 rig=bpy.data.objects[name+' | Motion rig'];items=[o for o in rig.children if o.type=='MESH'];head=bpy.data.objects[name+' | Independent head generator']
 # Apply thickness before export. Existing skin is retained.
 for ob in items:
  bpy.context.view_layer.objects.active=ob
  for mod in list(ob.modifiers):
   if mod.type=='SOLIDIFY' and ('hair' in ob.name.lower() or 'lock' in ob.name.lower()):ob.modifiers.remove(mod)
   elif mod.type!='ARMATURE':bpy.ops.object.modifier_apply(modifier=mod.name)
 offset=rig.location.x;rig.location.x=0;bpy.context.view_layer.update()
 # Independent face asset, with the existing face-anchor origin convention.
 face=[];center=Vector((0,-.175 if name=='AceHack' else -.185,(1.80 if name=='AceHack' else 1.72)-.025))
 for source in items:
  if not (source==head or ('hair' in source.name.lower() and 'Hair-free' not in source.name) or any(x in source.name for x in ['Rooted shoulder lock','Mouth cavity','Upper teeth','Tongue'])):continue
  ob=source.copy();ob.data=source.data.copy();scene.collection.objects.link(ob);world=source.matrix_world.copy();ob.parent=None;ob.matrix_world=world
  for mod in list(ob.modifiers):ob.modifiers.remove(mod)
  if ob.data.shape_keys:
   for key in ob.data.shape_keys.key_blocks:
    for v in key.data:v.co-=center
  else:
   for v in ob.data.vertices:v.co-=center
  face.append(ob)
 bpy.ops.object.select_all(action='DESELECT')
 for ob in face:ob.select_set(True)
 path=O/(name+'-face-v14.glb');bpy.ops.export_scene.gltf(filepath=str(path),use_selection=True,export_format='GLB',export_animations=False,export_skins=False,export_morph=True,export_morph_normal=False,export_morph_tangent=False,export_image_format='JPEG',export_jpeg_quality=92)
 rows.append({'file':path.name,'bytes':path.stat().st_size,'kind':'face','morphs':11})
 for ob in face:bpy.data.objects.remove(ob,do_unlink=True)
 # Bake an actual T-pose into geometry and rest skeleton for Body Avatar Drive.
 for side,sgn in [('Left',1),('Right',-1)]:
  for part in ['UpperArm','ForeArm','Hand']:
   bone=rig.pose.bones[side+part];m=bone.matrix.copy();pivot=m.translation.copy();direction=(bone.tail-bone.head).normalized();rot=direction.rotation_difference(Vector((sgn,0,0))).to_matrix().to_4x4();from mathutils import Matrix
   bone.matrix=Matrix.Translation(pivot)@rot@Matrix.Translation(-pivot)@m;bpy.context.view_layer.update()
 for ob in items:
  bpy.context.view_layer.objects.active=ob
  for mod in list(ob.modifiers):
   if mod.type=='ARMATURE':
    if ob.data.shape_keys:ob.modifiers.remove(mod)
    else:bpy.ops.object.modifier_apply(modifier=mod.name)
 bpy.ops.object.select_all(action='DESELECT');rig.select_set(True);bpy.context.view_layer.objects.active=rig;bpy.ops.object.mode_set(mode='POSE');bpy.ops.pose.armature_apply(selected=False);bpy.ops.object.mode_set(mode='OBJECT')
 for ob in items:
  mod=ob.modifiers.new('T-pose mobile skin','ARMATURE');mod.object=rig
 bpy.ops.object.select_all(action='DESELECT');rig.select_set(True)
 for ob in items:ob.select_set(True)
 path=O/(name+'-body-v14.glb');bpy.ops.export_scene.gltf(filepath=str(path),use_selection=True,export_format='GLB',export_animations=False,export_morph=True,export_morph_normal=False,export_morph_tangent=False,export_image_format='JPEG',export_jpeg_quality=92)
 rows.append({'file':path.name,'bytes':path.stat().st_size,'kind':'body','bones':len(rig.data.bones),'rest_pose':'T'})
 rig.location.x=offset
bpy.ops.wm.save_as_mainfile(filepath=str(O/'TikTok-Candidates-v14.blend'));(R/'mobile-export-v14.json').write_text(json.dumps(rows,indent=2))
