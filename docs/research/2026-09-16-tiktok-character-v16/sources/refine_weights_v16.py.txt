def array(collection,property,width,dtype=float):
 a=np.empty(len(collection)*width,dtype=dtype);collection.foreach_get(property,a);return a.reshape(-1,width)
def fingerprint(o):
 digest=hashlib.sha256()
 for data,prop,width,dtype in [(o.data.vertices,'co',3,np.float32),(o.data.loops,'vertex_index',1,np.int32),(o.data.polygons,'material_index',1,np.int32)]:digest.update(array(data,prop,width,dtype).tobytes())
 for uv in o.data.uv_layers:digest.update(array(uv.data,'uv',2,np.float32).tobytes())
 return digest.hexdigest()
base={o.name:array(o.data.vertices,'co',3) for o in meshes}
edges={o.name:array(o.data.edges,'vertices',2,np.int32) for o in meshes}

def pose(ops):
 for rig in rigs:
  for b in rig.pose.bones:b.matrix_basis=Matrix.Identity(4)
  bpy.context.view_layer.update()
  for name,axis,degrees in ops:
   b=rig.pose.bones[name];m=b.matrix.copy();p=m.translation.copy()
   b.matrix=Matrix.Translation(p)@Matrix.Rotation(math.radians(degrees),4,axis)@Matrix.Translation(-p)@m
   bpy.context.view_layer.update()
 bpy.context.view_layer.update()
poses=[('neutral',[]),('left-arm',[('LeftUpperArm','Y',-50)]),('right-arm',[('RightUpperArm','Y',50)]),('left-knee',[('LeftThigh','X',-30),('LeftShin','X',55)]),('right-knee',[('RightThigh','X',-30),('RightShin','X',55)]),('head-left',[('Head','Z',30)]),('head-right',[('Head','Z',-30)]),('both-arms-75',[('LeftUpperArm','Y',-75),('RightUpperArm','Y',75)]),('elbows',[('LeftForeArm','X',-60),('RightForeArm','X',-60)]),('combined',[('LeftUpperArm','Y',-50),('Head','Z',30),('RightThigh','X',-30),('RightShin','X',55)])]
def evaluate(stage):
 rows=[]
 for label,ops in poses:
  pose(ops);dg=bpy.context.evaluated_depsgraph_get()
  for o in meshes:
   ob=o.evaluated_get(dg);m=ob.to_mesh();v=array(m.vertices,'co',3);a=base[o.name];e=edges[o.name]
   assert len(v)==len(a) and np.isfinite(v).all()
   length=np.linalg.norm(a[e[:,0]]-a[e[:,1]],axis=1);current=np.linalg.norm(v[e[:,0]]-v[e[:,1]],axis=1)
   valid=length>.001;ratio=current[valid]/length[valid]
   rows.append({'stage':stage,'pose':label,'mesh':o.name,'edges_over_3x':int((ratio>3).sum()),'edges_under_one_third':int((ratio<1/3).sum()),'stretch_p99':float(np.quantile(ratio,.99)),'max_stretch':float(ratio.max()),'max_displacement_m':float(np.linalg.norm(v-a,axis=1).max())})
   ob.to_mesh_clear()
 print(stage, 'extreme stretched edges',sum(r['edges_over_3x'] for r in rows),flush=True)
 return rows
report=evaluate('v15')
pose([]);changes=[]
for character,scale in [('AceHack',1),('Xenaa',1.82/1.9)]:
 for suffix in ['Body generator','Independent head generator']:
  o=bpy.data.objects[character+' | '+suffix];xyz=base[o.name];e=edges[o.name];n=len(xyz)
  groups=[g for g in o.vertex_groups if any(vg.group==g.index and vg.weight>0 for v in o.data.vertices for vg in v.groups)]
  lookup={g.index:i for i,g in enumerate(groups)};w=np.zeros((n,len(groups)))
  for v in o.data.vertices:
   for g in v.groups:
    if g.group in lookup:w[v.index,lookup[g.group]]=g.weight
  if suffix=='Body generator':free=(xyz[:,2]>1.28*scale)&(xyz[:,2]<1.60*scale)&(np.abs(xyz[:,0])>.12*scale)
  else:
   free=(xyz[:,2]>(1.53 if character=='AceHack' else 1.485))&(xyz[:,2]<(1.66 if character=='AceHack' else 1.60))
   marker=o.data.attributes.get('reviewScalpHair')
   if marker:
    for i,d in enumerate(marker.data):
     if d.value:free[i]=False
  src=np.r_[e[:,0],e[:,1]];dst=np.r_[e[:,1],e[:,0]]
  use=free[src];src=src[use];dst=dst[use]
  # Positive inverse-length weights; fixed outside region, fidelity to original inside.
  length=np.linalg.norm(xyz[src]-xyz[dst],axis=1);weight=1/np.maximum(length,.001)
  degree=np.bincount(src,weights=weight,minlength=n);active=free&(degree>0);den=np.maximum(degree,1)
  updated=w.copy()
  for k in range(w.shape[1]):
   value=w[:,k].copy()
   for iteration in range(400):
    avg=np.bincount(src,weights=weight*value[dst],minlength=n)/den
    target=(avg+.015*w[:,k])/1.015
    value[active]=.2*value[active]+.8*target[active]
   updated[:,k]=value
  assert np.isfinite(updated).all() and updated.min()>-1e-8
  updated/=updated.sum(axis=1,keepdims=True)
  assert np.max(np.abs(updated[~free]-w[~free]))<1e-6
  # Apply only changed vertices. Mesh positions, UVs, materials and rig remain untouched.
  ids=np.where(np.max(np.abs(updated-w),axis=1)>1e-7)[0]
  for i in ids:
   for k,g in enumerate(groups):g.add([int(i)],float(updated[i,k]),'REPLACE')
  changes.append({'mesh':o.name,'changed_vertices':len(ids),'max_weight_delta':float(np.abs(updated-w).max()),'iterations':400,'fidelity':.015,'free_vertices':int(free.sum())})
  print(changes[-1],flush=True)
