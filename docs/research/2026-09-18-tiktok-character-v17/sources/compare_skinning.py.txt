"""Bounded, rest-surface-preserving skin-weight experiment; not a mobile asset."""
import bpy, hashlib, json, math
import numpy as np
from pathlib import Path
from mathutils import Matrix
R=Path(__file__).resolve().parent
SOURCE=R.parent/'TikTok-Character-Quality-v16/review/Blender-Quality-Review-v16.blend'
assert hashlib.sha256(SOURCE.read_bytes()).hexdigest()=='899cf39959455af43942c2163691775582bafdf0f1a971d1d8322cd79fb3d35f', 'Restore the exact archived v16 input'
O=R/'review';O.mkdir(exist_ok=True)
bpy.ops.wm.open_mainfile(filepath=str(SOURCE));sc=bpy.context.scene
rigs=[bpy.data.objects[n+' | Motion rig'] for n in ['AceHack','Xenaa']]
for rig in rigs:
 rig.animation_data_clear()
 for bone in rig.pose.bones:bone.matrix_basis=Matrix.Identity(4)
sc.frame_set(1);bpy.context.view_layer.update()
meshes=[o for o in sc.objects if o.type=='MESH' and any(m.type=='ARMATURE' for m in o.modifiers) and 'Collar' not in o.name]
def array(collection,property,width,dtype=float):
 a=np.empty(len(collection)*width,dtype=dtype);collection.foreach_get(property,a);return a.reshape(-1,width)
base={o.name:array(o.data.vertices,'co',3) for o in meshes}
edges={o.name:array(o.data.edges,'vertices',2,np.int32) for o in meshes}
def pose(ops):
 for rig in rigs:
  for b in rig.pose.bones:b.matrix_basis=Matrix.Identity(4)
  bpy.context.view_layer.update()
  for name,axis,degrees in ops:
   b=rig.pose.bones[name];m=b.matrix.copy();p=m.translation.copy()
   b.matrix=Matrix.Translation(p)@Matrix.Rotation(math.radians(degrees),4,axis)@Matrix.Translation(-p)@m
   bpy.context.view_layer.update()
 bpy.context.view_layer.update()
poses=[('neutral',[]),('left-arm',[('LeftUpperArm','Y',-50)]),('right-arm',[('RightUpperArm','Y',50)]),('left-knee',[('LeftThigh','X',-30),('LeftShin','X',55)]),('right-knee',[('RightThigh','X',-30),('RightShin','X',55)]),('head-left',[('Head','Z',30)]),('head-right',[('Head','Z',-30)]),('both-arms-75',[('LeftUpperArm','Y',-75),('RightUpperArm','Y',75)]),('elbows',[('LeftForeArm','X',-60),('RightForeArm','X',-60)]),('combined',[('LeftUpperArm','Y',-50),('Head','Z',30),('RightThigh','X',-30),('RightShin','X',55)])]
def evaluate(stage):
 rows=[]
 for label,ops in poses:
  pose(ops);dg=bpy.context.evaluated_depsgraph_get()
  for o in meshes:
   ob=o.evaluated_get(dg);m=ob.to_mesh();v=array(m.vertices,'co',3);a=base[o.name];e=edges[o.name]
   assert len(v)==len(a) and np.isfinite(v).all()
   length=np.linalg.norm(a[e[:,0]]-a[e[:,1]],axis=1);current=np.linalg.norm(v[e[:,0]]-v[e[:,1]],axis=1)
   valid=length>.001;ratio=current[valid]/length[valid]
   rows.append({'stage':stage,'pose':label,'mesh':o.name,'edges_over_3x':int((ratio>3).sum()),'edges_under_one_third':int((ratio<1/3).sum()),'stretch_p99':float(np.quantile(ratio,.99)),'max_stretch':float(ratio.max()),'max_displacement_m':float(np.linalg.norm(v-a,axis=1).max())})
   ob.to_mesh_clear()
 print(stage, 'extreme stretched edges',sum(r['edges_over_3x'] for r in rows),flush=True)
 return rows

report=evaluate('v16-linear')
for o in meshes:
 for m in o.modifiers:
  if m.type=='ARMATURE':m.use_deform_preserve_volume=True
report+=evaluate('v16-preserve-volume')
(R/'comparison.json').write_text(json.dumps(report,indent=2)+'\n')
pose([])
print('COMPARISON COMPLETE',flush=True)
