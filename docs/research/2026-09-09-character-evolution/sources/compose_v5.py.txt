"""Compose independently reconstructed heads and bodies in a review atelier."""
import bpy,bmesh,math,sys,json
from pathlib import Path
from mathutils import Vector,Matrix
root=Path(__file__).resolve().parent;lab=root.parent/'Character-Learning-Lab';sys.path.insert(0,str(root))
from review_heads import import_tsr,reference_material,bounds,studio
bpy.ops.wm.read_factory_settings(use_empty=True)
records=[]
for name,offset,target_h in [('AceHack',-.68,1.9),('Xenaa',.68,1.82)]:
 body=import_tsr(root/'fixed-latent'/name/'mesh-384.glb');body.name=name+' | Body generator'
 for p in body.data.polygons:p.use_smooth=True
 reference_material(body,lab/'references'/f'{name}-square-alpha.png')
 lo,hi=bounds(body);H=hi.z-lo.z;mid=(hi+lo)/2
 for v in body.data.vertices:v.co=Vector(((v.co.x-mid.x)*target_h/H,(v.co.y)*target_h/H,(v.co.z-lo.z)*target_h/H))
 body.data.update()
 # Remove the old low-detail central head; retain raised outer collar edges.
 bm=bmesh.new();bm.from_mesh(body.data)
 cut=[v for v in bm.verts if v.co.z>target_h*.905 or (v.co.z>target_h*.855 and abs(v.co.x)<target_h*.058)]
 bmesh.ops.delete(bm,geom=cut,context='VERTS');bm.to_mesh(body.data);bm.free()
 head=import_tsr(root/'heads'/name/'mesh.glb');head.name=name+' | Independent head generator'
 reference_material(head,root/'heads'/name/'alpha.png')
 lo,hi=bounds(head);H=hi.z-lo.z;mid=(hi+lo)/2;scale=target_h*(.205 if name=='AceHack' else .225)/H
 # Preserve UVs when composing; independent head field has much finer local detail.
 for v in head.data.vertices:v.co=Vector(((v.co.x-mid.x)*scale,v.co.y*scale-.008,(v.co.z-hi.z)*scale+target_h*1.022))
 head.data.update()
 # The hidden bust base can lie under the collar, but must not protrude through clothing.
 bm=bmesh.new();bm.from_mesh(head.data)
 cut=[v for v in bm.verts if v.co.z<target_h*.835 or (v.co.z<target_h*.864 and (v.co.x*v.co.x+(v.co.y+.008)**2)**.5>target_h*.045)]
 bmesh.ops.delete(bm,geom=cut,context='VERTS');bm.to_mesh(head.data);bm.free()
 # Repair only small open rings; large cut surfaces and garment boundaries remain explicit.
 bm=bmesh.new();bm.from_mesh(head.data)
 bmesh.ops.holes_fill(bm,edges=[e for e in bm.edges if e.is_boundary],sides=12)
 bm.to_mesh(head.data);bm.free()
 # Inferred back hair was pale. This declared material edit restores a dark rough hair surface.
 hair=bpy.data.materials.new(name+' | Dark back hair');hair.use_nodes=True
 n=hair.node_tree.nodes;l=hair.node_tree.links;bs=n.get('Principled BSDF')
 bs.inputs['Base Color'].default_value=(.025,.015,.009,1) if name=='AceHack' else (.009,.007,.012,1)
 bs.inputs['Roughness'].default_value=.72;bs.inputs['Specular IOR Level'].default_value=.18
 noise=n.new('ShaderNodeTexNoise');noise.inputs['Scale'].default_value=150;noise.inputs['Detail'].default_value=2
 bump=n.new('ShaderNodeBump');bump.inputs['Strength'].default_value=.16;bump.inputs['Distance'].default_value=.0015;l.new(noise.outputs['Fac'],bump.inputs['Height']);l.new(bump.outputs['Normal'],bs.inputs['Normal'])
 head.data.materials.append(hair)
 for p in head.data.polygons:
  c=p.center
  if (name=='Xenaa' and c.y>-.055) :p.material_index=len(head.data.materials)-1
 smooth=head.modifiers.new('Conservative head smoothing','SMOOTH');smooth.factor=.4;smooth.iterations=3
 # Feather a dark hair correction into the inferred rear scalp, avoiding a hard cap boundary.
 if name=='AceHack':
  attr=head.data.color_attributes.active_color
  if attr:
   for i,item in enumerate(attr.data):
    vi=head.data.loops[i].vertex_index if attr.domain=='CORNER' else i;c=head.data.vertices[vi].co
    w=max(0,min(1,(c.z-target_h*.918)/(target_h*.075)))*max(0,min(1,(c.y+.045)/.045))
    old=item.color_srgb;hair_color=(.18,.14,.115)
    item.color_srgb=tuple(old[j]*(1-w)+hair_color[j]*w for j in range(3))+(1,)
 # Explicit back-cloth hypothesis, separated from the measured front reference.
 mat=body.data.materials[0];n=mat.node_tree.nodes;l=mat.node_tree.links;bs=n.get('Principled BSDF');front=bs.inputs['Base Color'].links[0].from_socket
 geom=n.new('ShaderNodeNewGeometry');sep=n.new('ShaderNodeSeparateXYZ');l.new(geom.outputs['Normal'],sep.inputs[0])
 remap=n.new('ShaderNodeMapRange');remap.interpolation_type='SMOOTHSTEP';remap.inputs['From Min'].default_value=-.2;remap.inputs['From Max'].default_value=.5;l.new(sep.outputs['Y'],remap.inputs['Value'])
 mix=n.new('ShaderNodeMixRGB');l.new(remap.outputs[0],mix.inputs[0]);l.new(front,mix.inputs[1]);mix.inputs[2].default_value=(.012,.018,.013,1) if name=='AceHack' else (.035,.008,.062,1)
 if name=='Xenaa':
  position=n.new('ShaderNodeSeparateXYZ');l.new(geom.outputs['Position'],position.inputs[0])
  minus=n.new('ShaderNodeMath');minus.operation='SUBTRACT';l.new(position.outputs['X'],minus.inputs[0]);minus.inputs[1].default_value=offset
  absolute=n.new('ShaderNodeMath');absolute.operation='ABSOLUTE';l.new(minus.outputs[0],absolute.inputs[0])
  def cmp(socket,op,value):
   node=n.new('ShaderNodeMath');node.operation=op;l.new(socket,node.inputs[0]);node.inputs[1].default_value=value;return node.outputs[0]
  def mul(a,b):
   node=n.new('ShaderNodeMath');node.operation='MULTIPLY';l.new(a,node.inputs[0]);l.new(b,node.inputs[1]);return node.outputs[0]
  def region(base,color,factor):
   node=n.new('ShaderNodeMixRGB');l.new(factor,node.inputs[0]);l.new(base,node.inputs[1]);node.inputs[2].default_value=(*color,1);return node.outputs[0]
  base=n.new('ShaderNodeRGB');base.outputs[0].default_value=(.024,.006,.043,1);color=base.outputs[0]
  color=region(color,(.008,.01,.016),cmp(position.outputs['Z'],'LESS_THAN',target_h*.40))
  color=region(color,(.008,.007,.011),cmp(position.outputs['Z'],'GREATER_THAN',target_h*.73))
  arm=mul(cmp(absolute.outputs[0],'GREATER_THAN',target_h*.13),mul(cmp(position.outputs['Z'],'GREATER_THAN',target_h*.62),cmp(position.outputs['Z'],'LESS_THAN',target_h*.83)))
  color=region(color,(.40,.215,.145),arm)
  glove=mul(cmp(absolute.outputs[0],'GREATER_THAN',target_h*.15),cmp(position.outputs['Z'],'LESS_THAN',target_h*.64))
  color=region(color,(.008,.01,.014),glove);l.new(color,mix.inputs[2])
 l.new(mix.outputs[0],bs.inputs['Base Color']);l.new(mix.outputs[0],bs.inputs['Emission Color']);bs.inputs['Roughness'].default_value=.78;bs.inputs['Specular IOR Level'].default_value=.16
 # Suppress neutral-white hallucinations in inferred garment colors; front reference remains separate.
 attr=body.data.color_attributes.active_color
 if attr:
  for item in attr.data:
   c=item.color_srgb
   if min(c[:3])>.48 and max(c[:3])-min(c[:3])<.10:item.color_srgb=(.055,.05,.065,1)
 for o in [body,head]:o.location.x=offset
 # A small explicit collar joins the generators while retaining their separate editability.
 collar_mat=bpy.data.materials.new(name+' | Constructed collar');collar_mat.diffuse_color=(.55,.5,.4,1) if name=='AceHack' else (.025,.012,.045,1);collar_mat.use_nodes=True
 collar_mat.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=collar_mat.diffuse_color
 collar_mat.node_tree.nodes['Principled BSDF'].inputs['Roughness'].default_value=.78
 verts=[];faces=[];segments=48
 for j in range(segments+1):
  a=-math.pi/2+.58+j*(2*math.pi-1.16)/segments
  for upper in [0,1]:
   radius=target_h*(.040 if upper else .050)
   z=target_h*((.866 if name=='AceHack' else .874) if upper else .839)
   verts.append((offset+radius*math.cos(a),-.008+radius*.85*math.sin(a),z))
 for j in range(segments):faces.append((2*j,2*j+1,2*j+3,2*j+2))
 mesh=bpy.data.meshes.new(name+' collar mesh');mesh.from_pydata(verts,[],faces);mesh.update();collar=bpy.data.objects.new(name+' | Collar join',mesh);scene_collection=bpy.context.scene.collection;scene_collection.objects.link(collar);mesh.materials.append(collar_mat)
 for p in mesh.polygons:p.use_smooth=True
 sol=collar.modifiers.new('Cloth thickness','SOLIDIFY');sol.thickness=.002
 bevel=collar.modifiers.new('Soft seam','BEVEL');bevel.width=.0015;bevel.segments=2
 records.append(dict(name=name,height=target_h,body_vertices=len(body.data.vertices),head_vertices=len(head.data.vertices)))
scene,cam=studio(Vector((0,0,.99)),2.1)
scene.render.resolution_x=1800;scene.render.resolution_y=1500;cam.data.ortho_scale=2.85
# Floor, simple plinths, readable names; no effects hiding the meshes.
def mat(name,color,metal=0):
 m=bpy.data.materials.new(name);m.diffuse_color=(*color,1);m.use_nodes=True;bs=m.node_tree.nodes.get('Principled BSDF');bs.inputs['Base Color'].default_value=(*color,1);bs.inputs['Roughness'].default_value=.42;bs.inputs['Metallic'].default_value=metal;return m
floor=mat('Charcoal studio',(.018,.025,.04));gold=mat('Muted brass',(.4,.22,.065),.6);purple=mat('Violet titanium',(.17,.06,.3),.6)
bpy.ops.mesh.primitive_plane_add(size=200,location=(0,0,-.09));bpy.context.object.data.materials.append(floor)
for name,x,m in [('ACEHACK',-.68,gold),('XENAA',.68,purple)]:
 bpy.ops.mesh.primitive_cylinder_add(vertices=96,radius=.53,depth=.075,location=(x,0,-.045));o=bpy.context.object;o.name=name+' | Plinth';o.data.materials.append(floor)
 bevel=o.modifiers.new('Machined edge','BEVEL');bevel.width=.015;bevel.segments=3
 bpy.ops.mesh.primitive_torus_add(major_radius=.5,minor_radius=.005,major_segments=128,minor_segments=12,location=(x,0,-.007));bpy.context.object.data.materials.append(m)
 bpy.ops.object.text_add(location=(x,-.52,-.015),rotation=(math.pi/2,0,0));o=bpy.context.object;o.name=name+' | Label';o.data.body=name;o.data.align_x='CENTER';o.data.size=.055;o.data.extrude=.0006;o.data.materials.append(m)
out=root/'composed-v5';out.mkdir(exist_ok=True)
for a in [0,25,90,180]:
 r=math.radians(a);cam.location=Vector((4*math.sin(r),-4*math.cos(r),1.1));cam.rotation_euler=(Vector((0,0,.97))-cam.location).to_track_quat('-Z','Y').to_euler();scene.render.filepath=str(out/f'duo-{a}.png');bpy.ops.render.render(write_still=True)
cam.location=(.65,-5,1.3);cam.rotation_euler=(Vector((0,0,.96))-cam.location).to_track_quat('-Z','Y').to_euler()
for screen in bpy.data.screens:
 for area in screen.areas:
  if area.type=='VIEW_3D':area.spaces.active.region_3d.view_perspective='CAMERA';area.spaces.active.shading.type='MATERIAL'
scene['Evolution']='2026-09-09: composed independent head and body generators; provisional mesh study, not rigged.'
