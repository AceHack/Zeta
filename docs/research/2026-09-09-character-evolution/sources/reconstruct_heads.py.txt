"""Independent head generators using explicit U2Net preprocessing."""
import os
os.environ['PYTORCH_ENABLE_MPS_FALLBACK']='1'
from pathlib import Path
import sys,time,json,hashlib
import torch,numpy as np,rembg
from PIL import Image
root=Path(__file__).resolve().parent;lab=root.parent/'Character-Learning-Lab'
sys.path.insert(0,str(lab/'TripoSR'))
from tsr.system import TSR
from tsr.utils import remove_background,resize_foreground
torch.set_num_threads(12);torch.manual_seed(20260909)
model=TSR.from_pretrained(str(lab/'weights/TripoSR'),config_name='config.yaml',weight_name='model.ckpt').to('mps').eval()
model.renderer.set_chunk_size(16384)
session=rembg.new_session('u2net')
for name in ['AceHack','Xenaa']:
 out=root/'heads'/name;out.mkdir(parents=True,exist_ok=True)
 im=resize_foreground(remove_background(Image.open(root/'references'/f'{name}-head-derived.png'),session),.85)
 im.save(out/'alpha.png');rgba=np.array(im).astype(np.float32)/255
 conditioning=Image.fromarray(((rgba[:,:,:3]*rgba[:,:,3:4]+(1-rgba[:,:,3:4])*.5)*255).astype(np.uint8));conditioning.save(out/'conditioning.png')
 Image.fromarray(((rgba[:,:,:3]*rgba[:,:,3:4]+(1-rgba[:,:,3:4]))*255).astype(np.uint8)).save(out/'white.png')
 start=time.monotonic()
 with torch.no_grad():codes=model([conditioning],device='mps')
 torch.save(codes.detach().cpu(),out/'scene-code.pt')
 mesh=model.extract_mesh(codes,True,resolution=384)[0];mesh.export(out/'mesh.glb')
 row=dict(seconds=time.monotonic()-start,vertices=len(mesh.vertices),faces=len(mesh.faces),finite=bool(np.isfinite(mesh.vertices).all()),bounds=mesh.bounds.tolist())
 (out/'receipt.json').write_text(json.dumps(row,indent=2)+'\n');print(name,row,flush=True)
