bpy.ops.wm.open_mainfile(filepath=str(path));scene=bpy.context.scene
turntable=bpy.data.scenes.get('02 | Turntable')

   for curve in bag.fcurves:
    for key in curve.keyframe_points:key.interpolation='LINEAR'
