import os
os.environ['PYTORCH_ENABLE_MPS_FALLBACK']='1'
from pathlib import Path
import sys,time,json
import torch
from PIL import Image
root=Path(__file__).resolve().parent;sys.path.insert(0,str(root/'TripoSG'))
from triposg.pipelines.pipeline_triposg import TripoSGPipeline
print('Loading SG',flush=True)
torch.set_num_threads(12)
pipe=TripoSGPipeline.from_pretrained(str(root/'weights/TripoSG'),torch_dtype=torch.float16).to('mps')
for name,filename in [('Xenaa','Xenaa-white.png'),('AceHack','AceHack-derived.png')]:
 out=root/'reconstruction-sg'/name;out.mkdir(parents=True,exist_ok=True)
 start=time.monotonic()
 result=pipe(image=Image.open(root/'references'/filename).convert('RGB'),generator=torch.Generator(device='cpu').manual_seed(20260908),num_inference_steps=30,guidance_scale=7.0,dense_octree_depth=7,hierarchical_octree_depth=7,use_flash_decoder=False)
 torch.mps.synchronize();mesh=result.meshes[0];mesh.export(out/'mesh.glb')
 report={'seconds':time.monotonic()-start,'vertices':len(mesh.vertices),'faces':len(mesh.faces),'method':'TripoSG fp16 MPS, 30 steps, dense 128 cubed, non-flash CPU marching cubes','source_image':filename,'texture':'none'}
 (out/'receipt.json').write_text(json.dumps(report,indent=2)+'\n');print(name,report,flush=True)
