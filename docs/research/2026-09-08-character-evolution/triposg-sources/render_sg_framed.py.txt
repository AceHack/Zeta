"""Render real reconstructed meshes from four cameras; no 2D image substitution."""
import bpy,math,json
from pathlib import Path
from mathutils import Vector, Matrix
root=Path('/Users/acehack/Documents/Blender/Character-Learning-Lab')
for name in ['AceHack','Xenaa']:
 bpy.ops.wm.read_factory_settings(use_empty=True)
 bpy.ops.import_scene.gltf(filepath=str(root/'reconstruction-sg-framed'/name/'mesh.glb'))
 objs=[o for o in bpy.context.scene.objects if o.type=='MESH']
 # TripoSG uses model Y-up, which glTF import converts to Z-up.
 bpy.context.view_layer.update()
 points=[o.matrix_world@Vector(p) for o in objs for p in o.bound_box]
 lo=Vector([min(v[i] for v in points) for i in range(3)])
 hi=Vector([max(v[i] for v in points) for i in range(3)])
 center=(lo+hi)/2; height=hi.z-lo.z
 print(name,'bounds',list(lo),list(hi),flush=True)
 for o in objs:
  for p in o.data.polygons:p.use_smooth=True
  material=bpy.data.materials.new('Clay');material.diffuse_color=(.45,.45,.45,1);o.data.materials.clear();o.data.materials.append(material)
 scene=bpy.context.scene;scene.render.engine='CYCLES';scene.cycles.samples=40
 scene.cycles.use_denoising=True
 prefs=bpy.context.preferences.addons['cycles'].preferences;prefs.compute_device_type='METAL';prefs.get_devices()
 for d in prefs.devices:d.use=(d.type=='METAL')
 scene.cycles.device='GPU'
 scene.world=bpy.data.worlds.new('Neutral studio');scene.world.use_nodes=True
 scene.world.node_tree.nodes['Background'].inputs[0].default_value=(0.13,0.13,0.13,1)
 scene.world.node_tree.nodes['Background'].inputs[1].default_value=.65
 def area(label,loc,power,size):
  data=bpy.data.lights.new(label,'AREA');data.energy=power;data.shape='DISK';data.size=size
  o=bpy.data.objects.new(label,data);scene.collection.objects.link(o);o.location=loc
  o.rotation_euler=(center-o.location).to_track_quat('-Z','Y').to_euler()
 area('Key',(3,-4,4),350,4);area('Fill',(-3,1,3),200,3)
 data=bpy.data.cameras.new('Review');cam=bpy.data.objects.new('Review',data);scene.collection.objects.link(cam);scene.camera=cam
 data.type='ORTHO';data.ortho_scale=height*1.18
 scene.render.resolution_x=900;scene.render.resolution_y=1200;scene.render.resolution_percentage=100
 scene.view_settings.view_transform='Standard'
 out=root/'reconstruction-sg-framed'/name
 for angle in [0,90,180,270]:
  a=math.radians(angle);cam.location=center+Vector((4*math.cos(a),4*math.sin(a),.08))
  cam.rotation_euler=(center-cam.location).to_track_quat('-Z','Y').to_euler()
  scene.render.filepath=str(out/f'review-{angle}.png');bpy.ops.render.render(write_still=True)
 bpy.ops.wm.save_as_mainfile(filepath=str(out/f'{name}-reconstruction.blend'))
