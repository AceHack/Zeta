def curve(name,pts,r,m,closed=False,bone=None):
 d=bpy.data.curves.new(name,'CURVE');d.dimensions='3D';d.resolution_u=12;d.bevel_depth=r;d.bevel_resolution=3

def rod(name,a,b,r,m,bone=None,r2=None,verts=24):

def panel(name,rows,m,lining=None):
 # Each row is left/right; weave a mildly curved surface between seam lines.
 vs=[];N=8
 for a,b in rows:
  a,b=Vector(a),Vector(b)
  for j in range(N+1):
   t=j/N;p=a.lerp(b,t);p.y-=.006*math.sin(t*math.pi);vs.append(tuple(p))
 fs=[(i*(N+1)+j,i*(N+1)+j+1,(i+1)*(N+1)+j+1,(i+1)*(N+1)+j) for i in range(len(rows)-1) for j in range(N)]
 fs=[f if (Vector(vs[f[1]])-Vector(vs[f[0]])).cross(Vector(vs[f[2]])-Vector(vs[f[0]])).y<0 else tuple(reversed(f)) for f in fs]
 o=mesh(name,vs,fs,m,sub=2,solid=.003)
 if lining:
  o.data.materials.append(lining);o.modifiers[-1].material_offset=1
 return o
