def reg(o,name,m=None,bone=None,weighted=True):
 o.name=who+' | '+name
 for c in list(o.users_collection):c.objects.unlink(o)
 C.objects.link(o);o.parent=root;o['Character']=who
 if m and hasattr(o.data,'materials'):o.data.materials.append(m)
 if o.type=='MESH':
  for p in o.data.polygons:p.use_smooth=True
  if weighted:
   groups={}
   for v in o.data.vertices:
    if bone:weights=[(bone,1.0)]
    else:
     _,idx,_=kd.find(o.matrix_basis@v.co);weights=[(body.vertex_groups[g.group].name,g.weight) for g in body.data.vertices[idx].groups if body.vertex_groups[g.group].name in rig.data.bones]
    for n,w in weights:
     if n not in groups:groups[n]=o.vertex_groups.new(name=n)
     groups[n].add([v.index],w,'REPLACE')
   mod=o.modifiers.new('Character deformation','ARMATURE');mod.object=rig
 return o

def mesh(name,vs,fs,m,bone=None,sub=0,solid=0):
 d=bpy.data.meshes.new(name);d.from_pydata(vs,[],fs);d.update();o=bpy.data.objects.new(name,d);C.objects.link(o);reg(o,name,m,bone)
 if sub:md=o.modifiers.new('Tailored smooth surface','SUBSURF');md.levels=sub;md.render_levels=sub
 if solid:md=o.modifiers.new('Real fabric thickness','SOLIDIFY');md.thickness=solid
 return o

def curve(name,pts,r,m,closed=False,bone=None):
 d=bpy.data.curves.new(name,'CURVE');d.dimensions='3D';d.resolution_u=12;d.bevel_depth=r;d.bevel_resolution=3
 s=d.splines.new('BEZIER');s.bezier_points.add(len(pts)-1)
 for p,co in zip(s.bezier_points,pts):p.co=co;p.handle_left_type='AUTO';p.handle_right_type='AUTO'
 s.use_cyclic_u=closed;o=bpy.data.objects.new(name,d);C.objects.link(o)
 # Convert paths to weighted geometry so seams remain fitted while posing.
 bpy.ops.object.select_all(action='DESELECT');o.select_set(True);bpy.context.view_layer.objects.active=o;bpy.ops.object.convert(target='MESH');o=bpy.context.object
 return reg(o,name,m,bone)

def multi_hair(name,lines,r,m,bone='head'):
 d=bpy.data.curves.new(name,'CURVE');d.dimensions='3D';d.resolution_u=3;d.bevel_depth=r;d.bevel_resolution=1
 for pts in lines:
  s=d.splines.new('POLY');s.points.add(len(pts)-1)
  for j,(p,co) in enumerate(zip(s.points,pts)):p.co=(*co,1);p.radius=1-j/(len(pts)-1)*.83
 o=bpy.data.objects.new(name,d);C.objects.link(o);reg(o,name,m,weighted=False)
 # Curves bind to the head using a transform driver constraint, preserving rest placement.
 con=o.constraints.new('CHILD_OF');con.target=rig;con.subtarget=bone;con.inverse_matrix=(rig.matrix_world@rig.pose.bones[bone].matrix).inverted()@root.matrix_world
 return o

def sphere(name,p,scale,m,bone=None):
 bpy.ops.mesh.primitive_uv_sphere_add(segments=24,ring_count=12,location=p);o=bpy.context.object;o.scale=scale;bpy.ops.object.transform_apply(location=True,rotation=True,scale=True);return reg(o,name,m,bone)

def rod(name,a,b,r,m,bone=None,r2=None,verts=24):
 a,b=Vector(a),Vector(b);bpy.ops.mesh.primitive_cone_add(vertices=verts,radius1=r,radius2=r if r2 is None else r2,depth=(b-a).length,location=(a+b)/2);o=bpy.context.object;o.rotation_euler=(b-a).to_track_quat('Z','Y').to_euler();bpy.ops.object.transform_apply(location=True,rotation=True,scale=True);return reg(o,name,m,bone)

def buckle(name,p,w,h,m=brass):
 x,y,z=p;pts=[(x-w/2,y,z-h/2),(x+w/2,y,z-h/2),(x+w/2,y,z+h/2),(x-w/2,y,z+h/2)];curve(name,pts,.002,m,True);rod(name+' tongue',(x,y-.001,z),(x+w*.48,y-.001,z),.0012,m)

def ray(x,z,off=.001):

def panel(name,rows,m,lining=None):
 # Each row is left/right; weave a mildly curved surface between seam lines.
 vs=[];N=8
 for a,b in rows:
  a,b=Vector(a),Vector(b)
  for j in range(N+1):
   t=j/N;p=a.lerp(b,t);p.y-=.006*math.sin(t*math.pi);vs.append(tuple(p))
 fs=[(i*(N+1)+j,i*(N+1)+j+1,(i+1)*(N+1)+j+1,(i+1)*(N+1)+j) for i in range(len(rows)-1) for j in range(N)]
 o=mesh(name,vs,fs,m,sub=2,solid=.003)
 if lining:
  o.data.materials.append(lining);o.modifiers[-1].material_offset=1
 return o

def surface_shell(name,predicate,m,offset=.007,smooth=4):
 # Retains real anatomical quad topology and original skinning weights.
 bg=body.vertex_groups['body'].index;allowed={v.index for v in body.data.vertices if any(g.group==bg and g.weight>.5 for g in v.groups)}
 polys=[p for p in body.data.polygons if all(i in allowed for i in p.vertices) and predicate(sum((body.data.vertices[i].co for i in p.vertices),Vector())/len(p.vertices))]
 ids=sorted({i for p in polys for i in p.vertices});mp={a:i for i,a in enumerate(ids)}
 vs=[body.data.vertices[i].co+body.data.vertices[i].normal*offset for i in ids];fs=[tuple(mp[i] for i in p.vertices) for p in polys]
 o=mesh(name,vs,fs,m,sub=0)
 if smooth:md=o.modifiers.new('Garment relaxation','SMOOTH');md.factor=.9;md.iterations=smooth
 md=o.modifiers.new('Garment subdivision','SUBSURF');md.levels=2;md.render_levels=2
 md=o.modifiers.new('Garment thickness','SOLIDIFY');md.thickness=.004
 return o
