import bpy,json,math
from pathlib import Path
R=Path('/Users/acehack/Documents/Blender/AceHack-Xenaa-Realistic');bpy.ops.wm.open_mainfile(filepath=str(R/'AceHack_Xenaa_Reference_Revision.blend'))
(R/'exports').mkdir(exist_ok=True);report=[]
for who in ['AceHack','Xenaa']:
 c=bpy.data.collections[who+' | Character'];root=bpy.data.objects[who+' | Root'];x=root.location.x;root.location.x=0
 bpy.ops.object.select_all(action='DESELECT')
 visible=[o for o in c.objects if not o.hide_render or o.type in ['ARMATURE','EMPTY']]
 for o in visible:o.hide_set(False);o.hide_viewport=False;o.select_set(True)
 bpy.context.view_layer.update()
 out=R/'exports'/(who+'-rigged.glb')
 bpy.ops.export_scene.gltf(filepath=str(out),export_format='GLB',use_selection=True,use_active_scene=True,export_apply=True,export_animations=False,export_skins=True,export_rest_position_armature=True,export_yup=True,export_extras=True)
 report.append({'character':who,'glb':str(out.relative_to(R)),'selectedObjects':len(visible),'rigBones':len(bpy.data.objects[who+' | Deformation Rig'].data.bones),'bytes':out.stat().st_size,'pose':'rest pose for interchange; posed master is in the Blender scene'})
 root.location.x=x
(R/'exports/export-report.json').write_text(json.dumps(report,indent=2)+'\n')
print('EXPORT_COMPLETE',report)
