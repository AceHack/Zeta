def smooth(o):
 if hasattr(o.data,'polygons'):
  for p in o.data.polygons:p.use_smooth=True
 return o
def mesh(name,verts,faces,mat,sub=0,solid=0,bevel=0,bone=None):
 d=bpy.data.meshes.new(name);d.from_pydata(verts,[],faces);d.update();o=bpy.data.objects.new(name,d);COL.objects.link(o)
 if mat:d.materials.append(mat)
 if OWNER:o['Character']=OWNER;o['Part']=bone or PART;OWNERS[OWNER].append(o)
 smooth(o)
 if sub:mod=o.modifiers.new('Silhouette subdivision','SUBSURF');mod.levels=sub
 if solid:mod=o.modifiers.new('Tailored thickness','SOLIDIFY');mod.thickness=solid
 if bevel:mod=o.modifiers.new('Machined edge catches','BEVEL');mod.width=bevel;mod.segments=3
 return o

def uv(name,loc,scale,mat,bone=None,seg=40,rings=24):

def cube(name,loc,scale,mat,bev=.015,bone=None):

 if bev:m=o.modifiers.new('Rounded crafted edges','BEVEL');m.width=bev;m.segments=3;o.modifiers.new('Weighted normals','WEIGHTED_NORMAL')

def cyl(name,loc,r,depth,mat,axis=(0,0,1),verts=48,bone=None):
 bpy.ops.mesh.primitive_cylinder_add(vertices=verts,radius=r,depth=depth,location=loc);o=attach(bpy.context.object,name,mat,bone);o.rotation_mode='QUATERNION';o.rotation_quaternion=Vector(axis).to_track_quat('Z','Y')

def curve(name,pts,mat,r=.012,cyclic=False,radii=None,bone=None):
 d=bpy.data.curves.new(name,'CURVE');d.dimensions='3D';d.resolution_u=12;d.bevel_depth=r;d.bevel_resolution=3
 s=d.splines.new('BEZIER');s.bezier_points.add(len(pts)-1);s.use_cyclic_u=cyclic
 for i,(p,co) in enumerate(zip(s.bezier_points,pts)):
  p.co=co;p.handle_left_type='AUTO';p.handle_right_type='AUTO';p.radius=radii[i] if radii else 1
 o=bpy.data.objects.new(name,d);COL.objects.link(o);d.materials.append(mat)
 if OWNER:o['Character']=OWNER;o['Part']=bone or PART;OWNERS[OWNER].append(o)
 return o

def limb(name,a,b,ra,rb,mat,bone):
