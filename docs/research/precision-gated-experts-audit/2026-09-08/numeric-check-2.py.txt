# Independent deterministic quadrature of an explicit synthetic scalar density.
# No upstream code, data, model or training is invoked.
def simpson(power,n):
 lo,hi=-12.,6.;h=(hi-lo)/n
 def f(z): return math.exp(-.5*z*z-math.exp(z)+power*z)
 return h/3*(f(lo)+f(hi)+sum((4 if k%2 else 2)*f(lo+k*h) for k in range(1,n)))
