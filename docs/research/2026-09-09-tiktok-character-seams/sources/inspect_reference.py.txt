import bpy,json,math
from pathlib import Path
root=Path(__file__).resolve().parent
scene=bpy.context.scene
meshes=[o for o in scene.objects if o.type=='MESH' and any(k in o.name for k in ['generator','Fitted collar','Rebuilt hair'])]
assert len(meshes)==7
assert all(math.isfinite(c) for o in meshes for v in o.data.vertices for c in v.co)
images=[i for i in bpy.data.images if i.source=='FILE'];assert all(i.packed_file for i in images)
assert scene.camera is not None
rows=[]
for name in ['AceHack','Xenaa']:
 head=bpy.data.objects[name+' | Independent head generator'];body=bpy.data.objects[name+' | Body generator']
 hz=[(head.matrix_world@v.co).z for v in head.data.vertices];bz=[(body.matrix_world@v.co).z for v in body.data.vertices]
 assert min(hz)<max(bz)
 rows.append({'name':name,'head_z_bounds':[min(hz),max(hz)],'body_z_bounds':[min(bz),max(bz)],'note':'Z-range overlap only; side/back renders remain necessary to assess attachment'})
(root/'reference-validation.json').write_text(json.dumps({'blender':bpy.app.version_string,'finite_vertices':True,'packed_file_images':[i.name for i in images],'character_meshes':len(meshes),'ranges':rows,'scope':'Neutral assembly structural checks, not whole-body watertightness or motion proof'},indent=2))
