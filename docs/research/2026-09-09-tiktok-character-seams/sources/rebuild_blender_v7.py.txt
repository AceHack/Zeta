def cut_part(source,label,predicate,capmat):
 # Duplicate topology/UVs, then complementary face selection: no stretched shared faces.
 part=source.copy();part.data=source.data.copy();scene.collection.objects.link(part);part.name=label
 bm=bmesh.new();bm.from_mesh(source.data);selected={f.index for f in bm.faces if predicate(f.calc_center_median())};bm.free()
 for o,keep in [(part,True),(source,False)]:
  bm=bmesh.new();bm.from_mesh(o.data);bmesh.ops.delete(bm,geom=[f for f in bm.faces if (f.index in selected)!=keep],context='FACES')
  bmesh.ops.delete(bm,geom=[v for v in bm.verts if not v.link_faces],context='VERTS')
  o.data.materials.append(capmat);idx=len(o.data.materials)-1
  edges=[e for e in bm.edges if e.is_boundary]
  # Cap only this inferred separation. The neck opening is handled separately.
  result=bmesh.ops.holes_fill(bm,edges=edges,sides=0)
  for f in result.get('faces',[]):f.material_index=idx
  finish(o,bm)
